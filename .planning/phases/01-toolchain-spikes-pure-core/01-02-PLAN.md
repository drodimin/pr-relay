```go
package model // internal/model — no imports beyond stdlib (fmt, net/url, regexp, strconv, strings)

type PRRef struct { Owner, Repo string; Number int }
func (r PRRef) String() string // "owner/repo#N"
func (r PRRef) URL() string    // "https://github.com/owner/repo/pull/N"
func (r PRRef) Valid() bool    // Owner and Repo match ^[A-Za-z0-9._-]{1,100}$ and Number > 0
func ParsePRURL(u string) (PRRef, error) // accepts only https://github.com/{o}/{r}/pull/{n} (optional trailing slash); returns Valid ref or error

type Actor struct { Login string; Type string } // Type = GraphQL __typename: "User","Bot","Mannequin","Organization", or "Ghost" (null author), "" (unknown commit author)
func (a Actor) IsUser() bool // Type == "User"

type CheckRun struct {
    Key string            // Workflow+"/"+Name when Workflow != "", else App+"/"+Name
    Name, Workflow, App string
    DatabaseID int64
    Status string         // CheckStatusState verbatim: COMPLETED|IN_PROGRESS|PENDING|QUEUED|REQUESTED|WAITING
    Conclusion string     // CheckConclusionState verbatim, "" until COMPLETED
    URL, DetailsURL string
    RunAttempt int
}
type ReviewComment struct { ID, URL, Path, ReplyToID string; Author Actor }
type Review struct { ID, State, URL string; Author Actor; HasBody bool; Comments []ReviewComment; CommentsTotal int }
type IssueComment struct { ID, URL string; Author Actor }
type Commit struct { OID string; Author Actor } // author.user.login, else committer.user.login, else Actor{} (unknown)
type ForcePush struct { ID, BeforeOID, AfterOID string; Actor Actor }

type Snapshot struct {
    Ref PRRef; NodeID string
    State string          // OPEN|CLOSED|MERGED
    MergedBy, ClosedBy Actor
    HeadOID, BaseOID string
    Mergeable string      // MERGEABLE|CONFLICTING|UNKNOWN
    Commits []Commit; CommitsTotal int          // oldest -> newest, as returned by commits(last:30)
    CIKnown bool                                // true when rollup is null or rollup.commit.oid == HeadOID
    Checks []CheckRun                           // CheckRun only, newest DatabaseID per Key, sorted by Key
    ChecksTruncated bool                        // contexts.pageInfo.hasNextPage
    UnknownConclusions []string                 // conclusion values outside the 9 known enums
    Reviews []Review; ReviewsTotal int
    Comments []IssueComment; CommentsTotal int
    ForcePushes []ForcePush
}

type EventType string
const (
    EventClosed    EventType = "closed"
    EventConflict  EventType = "conflict"
    EventCommits   EventType = "commits"
    EventCIFailed  EventType = "ci.failed"
    EventCISettled EventType = "ci.settled"
    EventReview    EventType = "review"
    EventComment   EventType = "comment"
)
func AllEventTypes() []EventType // exactly the order above (also the in-step emission order)
func (t EventType) Rank() int    // index in AllEventTypes, 99 if unknown

type Event struct {
    Type EventType; Key string // stable dedupe key, see 01-04
    Actor Actor; URL string
    CheckName, CheckWorkflow, CheckURL, HeadOID string // ci.failed (HeadOID also on ci.settled, commits)
    Passed bool; Total int; FailedNames []string         // ci.settled
    Verdict string; InlineCount int                      // review: Verdict is the raw GraphQL state (APPROVED|CHANGES_REQUESTED|COMMENTED|DISMISSED)
    CommentKind string; Path string                      // comment: "comment" (issue) | "reply" (thread reply) | "inline" (non-reply inline on an already-seen review)
    OldHeadOID string; Count int; ForcePushed bool; Authors []Actor // commits: Count 0 = unknown
    Conflicting bool                                     // conflict: true = became conflicting, false = cleared
    Merged bool                                          // closed
}

type Memory struct {
    Baselined bool
    HeadOID, BaseOID, State, Mergeable string // Mergeable: last RESOLVED value, "" = never resolved
    FailedReported map[string]bool            // CheckRun.Key -> ci.failed emitted for HeadOID
    SettleCandidate string                    // fingerprint of all-complete check set seen last step, "" none
    SettledResult string                      // "" | "passed" | "failed:<sorted,comma,joined,keys>" last reported for HeadOID
    SettleSeq int
    SeenReviews, SeenComments, SeenForcePushes map[string]bool
    ReviewsTotal, CommentsTotal int
}
func (m Memory) Clone() Memory // deep copy of every map; nil maps become empty maps
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub GraphQL response → Decode | All strings (check names, logins, paths, URLs) are attacker-influenced |
| capture.sh → GitHub | Runs gh with the user's token. Must only touch the dummy repo |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-04 | Information Disclosure | internal/ghquery/pr.graphql | mitigate | The query selects no comment bodies, and the review body only for HasBody. TestQueryShape asserts that `body` appears exactly once |
| T-01-05 | Information Disclosure | Decode | mitigate | The body is discarded after HasBody. TestDecodeDropsBodies asserts the sentinel is absent from the marshalled Snapshot |
| T-01-06 | Tampering | Decode (attacker strings) | transfer | Decode stores raw strings. Neutralization is the job of internal/sanitize (01-03) at render time, and render tests enforce it. Decode itself never formats output |
| T-01-07 | Spoofing | model.ParsePRURL | mitigate | Strict https + exact host github.com + no userinfo/port + charset-validated owner/repo. Table test with at least 10 rejections |
| T-01-08 | Elevation of Privilege | scripts/capture.sh | mitigate | The PR_RELAY_FIXTURE_REPO guard refuses any repo other than the declared dummy. Read-only GraphQL query only |
| T-01-09 | Information Disclosure | fixtures committed to a public repo | mitigate | capture.sh rewrites non-empty review bodies to PRRELAY-BODY-SENTINEL before writing to disk |
</threat_model>
