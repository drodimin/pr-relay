```go
type Snapshot struct {
    Ref PRRef; NodeID string; State string; MergedBy, ClosedBy Actor
    HeadOID, BaseOID string; Mergeable string
    Commits []Commit; CommitsTotal int; CIKnown bool; Checks []CheckRun; ChecksTruncated bool
    UnknownConclusions []string
    Reviews []Review; ReviewsTotal int; Comments []IssueComment; CommentsTotal int
    ForcePushes []ForcePush
}
type CheckRun struct { Key, Name, Workflow, App string; DatabaseID int64; Status, Conclusion, URL, DetailsURL string; RunAttempt int }
type Commit struct { OID string; Author Actor }
type ForcePush struct { ID, BeforeOID, AfterOID string; Actor Actor }
type Event struct {
    Type EventType; Key string; Actor Actor; URL string
    CheckName, CheckWorkflow, CheckURL, HeadOID string
    Passed bool; Total int; FailedNames []string
    Verdict string; InlineCount int; CommentKind string; Path string
    OldHeadOID string; Count int; ForcePushed bool; Authors []Actor
    Conflicting bool; Merged bool
}
type Memory struct {
    Baselined bool; HeadOID, BaseOID, State, Mergeable string
    FailedReported map[string]bool; SettleCandidate, SettledResult string; SettleSeq int
    SeenReviews, SeenComments, SeenForcePushes map[string]bool; ReviewsTotal, CommentsTotal int
}
func (m Memory) Clone() Memory
func (t EventType) Rank() int
```
