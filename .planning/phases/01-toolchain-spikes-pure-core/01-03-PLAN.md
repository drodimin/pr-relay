```go
package sanitize // stdlib only: strings, unicode, unicode/utf8, net/url, regexp
const (
    MaxLogin     = 40   // GitHub max login is 39
    MaxCheckName = 80
    MaxPath      = 100
    MaxURLBytes  = 300
)
func Field(s string, maxRunes int) string // single-line, stripped, collapsed, capped, tag-neutralized; "" -> "?"
func Login(s string) string               // Field(s, MaxLogin)
func Path(s string, maxRunes int) string   // like Field but keeps the TAIL: "…" + last (maxRunes-1) runes
func NeutralizeTag(s string) string        // case-insensitive "[pr-relay]" -> "(pr-relay)"
func URL(raw, fallback string) string      // raw if allowed, else fallback
func ShortSHA(s string) string             // 7 lowercase hex chars or "?"
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-10 | Tampering | sanitize.Field (terminal escape CSI/OSC/OSC52) | mitigate | Strip all !IsPrint runes, including C0/C1. Corpus entries \x1b[2J, \x1b]0;, \x1b]52;, \u009b. TestHostile property "every rune IsPrint or space" |
| T-01-11 | Elevation of Privilege | sanitize.Field (prompt injection by newline, early submit) | mitigate | Collapse every unicode.IsSpace rune to a single ' '. Property: no \n or \r |
| T-01-12 | Spoofing | sanitize.Field (fake `[pr-relay]` envelope) | mitigate | NeutralizeTag after stripping, so invisible-char splitting cannot rebuild the tag. Property: no case-insensitive [pr-relay] |
| T-01-13 | Tampering | sanitize.Field (invisible text smuggling: tags, VS, bidi, ZW) | mitigate | Strip Cf + Variation_Selector. Corpus entries for each |
| T-01-14 | Denial of Service | sanitize.Field (10 KB / combining flood) | mitigate | Rune cap with ellipsis. At most 2 combining marks per base |
| T-01-15 | Spoofing | sanitize.URL (malicious detailsUrl) | mitigate | Exact https://github.com host, no userinfo or port, printable ASCII, <=300 bytes. Otherwise a constructed fallback. Never truncate |
